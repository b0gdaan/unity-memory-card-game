How It Works
Game Mechanics
🔀

Fisher-Yates Shuffle

18 pairs of emoji icons randomly shuffled each game using the same algorithm as the Unity implementation.

⏱️

0.5s Mismatch Delay

When two non-matching cards are revealed, they stay visible briefly before flipping back — matching Unity's WaitForSeconds coroutine.

🏆

Score + Attempts

Score counts matched pairs, attempts count every flip pair. Replicated from Unity's TextMesh counters.

🔄

Instant Restart

New game reshuffles all cards and resets counters — equivalent to SceneManager.LoadScene in Unity.

Core Algorithm

The heart of the game — shuffle, grid generation, and coroutine-based match checking.

GameControllerScript.cs
// Fisher-Yates shuffle — same as Array.shuffle in JS
private int[] Randomiser(int[] locations) {
    int[] array = locations.Clone() as int[];
    for (int i = 0; i < array.Length; i++) {
        int j = Random.Range(i, array.Length);
        (array[i], array[j]) = (array[j], array[i]);
    }
    return array;
}

// Coroutine: wait 0.5s then flip unmatched cards back
private IEnumerator CheckGuessed() {
    if (firstOpen.spriteId == secondOpen.spriteId) {
        score++;
        scoreText.text = "Score: " + score;
    } else {
        yield return new WaitForSeconds(0.5f);
        firstOpen.Close();
        secondOpen.Close();
    }
    attempts++;
    attemptsText.text = "Attempts: " + attempts;
    firstOpen = null;
    secondOpen = null;
}

// Grid generation: 6×6, position via Xspace/Yspace constants
private void Start() {
    int[] locations = { 0,0,1,1,2,2,...17,17 };
    locations = Randomiser(locations);
    for (int i = 0; i < columns; i++)
        for (int j = 0; j < rows; j++)
            SpawnCard(i, j, locations[j * columns + i]);
}
